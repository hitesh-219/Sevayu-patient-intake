    'United States': ['Alabama','Alaska','Arizona','Arkansas','California','Colorado','Connecticut','Delaware','District of Columbia','Florida','Georgia','Hawaii','Idaho','Illinois','Indiana','Iowa','Kansas','Kentucky','Louisiana','Maine','Maryland','Massachusetts','Michigan','Minnesota','Mississippi','Missouri','Montana','Nebraska','Nevada','New Hampshire','New Jersey','New Mexico','New York','North Carolina','North Dakota','Ohio','Oklahoma','Oregon','Pennsylvania','Rhode Island','South Carolina','South Dakota','Tennessee','Texas','Utah','Vermont','Virginia','Washington','West Virginia','Wisconsin','Wyoming']

  };

 

  var form    = document.getElementById('intake');

  var country = document.getElementById('country');

  var region  = document.getElementById('region');

 

  /* Build the 0–10 pain scale */

  var track = document.getElementById('painTrack');

  for (var i = 0; i <= 10; i++) {

    var r = document.createElement('input');

    r.type = 'radio'; r.name = 'pain'; r.value = String(i); r.id = 'pain' + i;

    if (i === 0) r.setAttribute('data-req', '');

    var l = document.createElement('label');

    l.setAttribute('for', 'pain' + i);

    l.textContent = i;

    track.appendChild(r); track.appendChild(l);

  }

 

  /* Country drives the region list and the postal code label */

  country.addEventListener('change', function () {

    var list = REGIONS[country.value] || [];

    region.innerHTML = '';

    var ph = document.createElement('option');

    ph.value = '';

    ph.textContent = list.length ? 'Select…' : 'Select a country first…';

    region.appendChild(ph);

    list.forEach(function (name) {

      var o = document.createElement('option');

      o.value = name; o.textContent = name;

      region.appendChild(o);

    });

    var us = country.value === 'United States';

    document.getElementById('regionLabel').textContent = us ? 'State' : 'Province / Territory';

    document.getElementById('postalLabel').textContent = us ? 'ZIP code' : 'Postal code';

    document.getElementById('postal').setAttribute('placeholder', us ? '90210' : 'M5V 2T6');

  });

 

  /* ---- validation ---- */

 

  function fieldOf(el) { return el.closest('.field'); }

 

  function painPicked() {

    return !!form.querySelector('input[name="pain"]:checked');

  }

 

  function consentDone() {

    return ['cAccurate','cTreat','cPrivacy'].every(function (id) {

      return document.getElementById(id).checked;

    });

  }

 

  function valid(el) {

    var v = (el.value || '').trim();

    if (el.type === 'checkbox') return el.checked;

    if (!v) return false;

    if (el.dataset.type === 'email') return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v);

    if (el.dataset.type === 'phone') return (v.replace(/\D/g, '').length >= 10);

    return true;

  }

 

  /* Every required control, with pain and consent counted as one item each */

  var required = Array.prototype.slice.call(

    form.querySelectorAll('[data-req]')

  ).filter(function (el) { return el.name !== 'pain'; });

 

  var textRequired = required.filter(function (el) { return el.type !== 'checkbox'; });

 

  function tally() {

    var done = textRequired.filter(valid).length;

    if (painPicked()) done++;

    if (consentDone()) done++;

    var total = textRequired.length + 2;

    document.getElementById('mDone').textContent = done;

    document.getElementById('mTotal').textContent = total;

    document.getElementById('mFill').style.width = Math.round((done / total) * 100) + '%';

  }

 

  form.addEventListener('input', tally);

  form.addEventListener('change', tally);

 

  textRequired.forEach(function (el) {

    el.addEventListener('blur', function () {

      fieldOf(el).classList.toggle('bad', !valid(el));

    });

    el.addEventListener('input', function () {

      if (valid(el)) fieldOf(el).classList.remove('bad');

    });

  });

 

  /* ---- submit ---- */

 

  form.addEventListener('submit', function (e) {

    e.preventDefault();

 

    var bad = null;

    textRequired.forEach(function (el) {

      var ok = valid(el);

      fieldOf(el).classList.toggle('bad', !ok);

      if (!ok && !bad) bad = fieldOf(el);

    });

 

    var painField = document.getElementById('painLabel').closest('.field');

    painField.classList.toggle('bad', !painPicked());

    if (!painPicked() && !bad) bad = painField;

 

    var cField = document.getElementById('consentField');

    cField.classList.toggle('bad', !consentDone());

    if (!consentDone() && !bad) bad = cField;

 

    var errBox = document.getElementById('formErr');

 

    if (bad) {

      errBox.textContent = 'A few answers still need filling in. They\'re marked below.';

      errBox.classList.add('on');

      bad.scrollIntoView({ behavior: 'smooth', block: 'center' });

      var focusable = bad.querySelector('input, select, textarea');

      if (focusable) focusable.focus({ preventScroll: true });

      return;

    }

 

    errBox.classList.remove('on');

 

    var btn = document.getElementById('sendBtn');

    btn.disabled = true;

    btn.textContent = 'Sending…';

 

    var payload = {};

    ['firstName','lastName','dob','pronouns','email','mobile','address','city','region',

     'country','postal','reminder','reason','onset','priorCare','conditions','medications',

     'allergies','surgeries','pregnant','focus','ecName','ecPhone','heard','website']

      .forEach(function (k) {

        var el = document.getElementById(k);

        payload[k] = el ? el.value.trim() : '';

      });

 

    payload.pain = form.querySelector('input[name="pain"]:checked').value;

    payload.consent = 'Accuracy, treatment, and privacy all agreed';


    // Submit data directly to Apps Script via fetch API

    fetch(SCRIPT_URL, {

      method: "POST",

      mode: "no-cors",

      headers: { "Content-Type": "text/plain" },

      body: JSON.stringify(payload)

    })

    .then(function () {

      document.getElementById('live').style.display = 'none';

      document.getElementById('done').classList.add('on');

      window.scrollTo(0, 0);

    })

    .catch(function (err) {

      fail(err && err.message);

    });

 

    function fail(msg) {

      btn.disabled = false;

      btn.textContent = 'Send my intake form';

      errBox.textContent = (msg || 'That didn\'t send.') + ' Check your connection and try again, or call the clinic and we\'ll take it over the phone.';

      errBox.classList.add('on');

      errBox.scrollIntoView({ behavior: 'smooth', block: 'center' });

    }

  });

 

  tally();

})();

</script>
